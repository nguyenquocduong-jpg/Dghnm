<script>
    async function loadData() {
        try {
            let response = await fetch('../results/case3.csv');
            if (!response.ok) response = await fetch('/results/case3.csv');
            if (!response.ok) response = await fetch('results/case3.csv');
            
            const text = await response.text();
            const lines = text.trim().split('\n').slice(1);
            
            const labels = [];
            const h2Gp = [], h3Gp = [];
            const h2Lat = [], h3Lat = [];

            lines.forEach(line => {
                const cols = line.split(',');
                if (cols.length >= 5) {
                    const time = cols[0].trim() + 's';
                    labels.push(time);
                    
                    h2Gp.push(parseFloat(cols[1]) || 0);
                    h2Lat.push(parseFloat(cols[2]) || 0);
                    h3Gp.push(parseFloat(cols[3]) || 0);
                    h3Lat.push(parseFloat(cols[4]) || 0);
                }
            });

            // 1. Goodput Chart
            new Chart(document.getElementById('goodputChart'), {
                type: 'line',
                data: {
                    labels: labels,
                    datasets: [
                        { label: 'HTTP/2 (TCP) Goodput (Mbps)', data: h2Gp, borderColor: '#e53e3e', borderWidth: 2, fill: false, pointRadius: 3 },
                        { label: 'HTTP/3 (QUIC) Goodput (Mbps)', data: h3Gp, borderColor: '#38a169', backgroundColor: 'rgba(56, 161, 105, 0.1)', fill: true, borderWidth: 2, pointRadius: 3 }
                    ]
                },
                options: { responsive: true, scales: { y: { min: 0, max: 110 } } }
            });

            // 2. Latency Spike Chart
            new Chart(document.getElementById('latencyChart'), {
                type: 'line',
                data: {
                    labels: labels,
                    datasets: [
                        { label: 'HTTP/2 Latency Spike (HoL Blocking - ms)', data: h2Lat, borderColor: '#e53e3e', backgroundColor: 'rgba(229, 62, 62, 0.1)', fill: true, borderWidth: 2, pointRadius: 4 },
                        { label: 'HTTP/3 Latency (Stable - ms)', data: h3Lat, borderColor: '#3182ce', borderWidth: 2, fill: false, pointRadius: 3 }
                    ]
                },
                options: { responsive: true, scales: { y: { min: 0 } } }
            });

        } catch (error) {
            console.error('Lỗi khi tải biểu đồ Case 3:', error);
        }
    }

    loadData();
</script>
